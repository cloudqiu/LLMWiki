我会给两个版本：

- 方案 A：用 
"perks/quantile" 的 
"Merge"（简单但有误差）
- 方案 B：用 
"t-digest" 的 
"Merge"（推荐，合并精度更好）

方案 A：perks/quantile 多实例合并

package main

import (
	"fmt"
	"math/rand"
	"sync"
	"time"

	"github.com/beorn7/perks/quantile"
)

// 模拟一个服务实例
type Instance struct {
	id     int
	stream *quantile.Stream
	count  int
}

func (inst *Instance) handleRequests(done <-chan struct{}, wg *sync.WaitGroup) {
	defer wg.Done()
	for {
		select {
		case <-done:
			return
		default:
			// 90% 快请求 10-50ms，10% 慢请求 100-300ms
			var latencyMs float64
			if rand.Float64() < 0.9 {
				latencyMs = 10 + rand.Float64()*40
			} else {
				latencyMs = 100 + rand.Float64()*200
			}
			inst.stream.Insert(latencyMs)
			inst.count++
			time.Sleep(time.Duration(5+rand.Intn(10)) * time.Millisecond)
		}
	}
}

func main() {
	rand.Seed(time.Now().UnixNano())

	// 创建 3 个实例
	instances := make([]*Instance, 3)
	for i := 0; i < 3; i++ {
		instances[i] = &Instance{
			id:     i,
			stream: quantile.NewTargeted(map[float64]float64{0.99: 0.001}),
		}
	}

	// 启动所有实例
	done := make(chan struct{})
	var wg sync.WaitGroup
	for _, inst := range instances {
		wg.Add(1)
		go inst.handleRequests(done, &wg)
	}

	// 运行 3 秒
	time.Sleep(3 * time.Second)
	close(done)
	wg.Wait()

	// ---- 中心节点：合并所有实例的 Stream ----
	globalStream := quantile.NewTargeted(map[float64]float64{0.99: 0.001})
	totalCount := 0
	for _, inst := range instances {
		globalStream.Merge(inst.stream.Samples())
		totalCount += inst.count
		fmt.Printf("  Pod %d: %d requests, local P99=%.1fms\n",
			inst.id, inst.count, inst.stream.Query(0.99))
	}

	fmt.Printf("\n总请求数: %d\n", totalCount)
	fmt.Printf("🌐 perks 合并后 全局 P99: %.1fms\n", globalStream.Query(0.99))
	fmt.Printf("🌐 全局 P90: %.1fms\n", globalStream.Query(0.90))
	fmt.Printf("🌐 全局 P50: %.1fms\n", globalStream.Query(0.50))
}

方案 B：t-digest 多实例合并（推荐）

t-digest 的合并是算法层面设计好的，合并后误差仍然可控，适合生产环境。

package main

import (
	"fmt"
	"math/rand"
	"sync"
	"time"

	"github.com/caio/go-tdigest"
)

type InstanceTD struct {
	id int
	td *tdigest.TDigest
}

func (inst *InstanceTD) handleRequests(done <-chan struct{}, wg *sync.WaitGroup) {
	defer wg.Done()
	for {
		select {
		case <-done:
			return
		default:
			var latencyMs float64
			if rand.Float64() < 0.9 {
				latencyMs = 10 + rand.Float64()*40
			} else {
				latencyMs = 100 + rand.Float64()*200
			}
			inst.td.Add(latencyMs)
			time.Sleep(time.Duration(5+rand.Intn(10)) * time.Millisecond)
		}
	}
}

func main() {
	rand.Seed(time.Now().UnixNano())

	// 创建 3 个实例，compression 越高精度越好（默认 100）
	instances := make([]*InstanceTD, 3)
	for i := 0; i < 3; i++ {
		instances[i] = &InstanceTD{
			id: i,
			td: tdigest.NewWithCompression(100),
		}
	}

	done := make(chan struct{})
	var wg sync.WaitGroup
	for _, inst := range instances {
		wg.Add(1)
		go inst.handleRequests(done, &wg)
	}

	time.Sleep(3 * time.Second)
	close(done)
	wg.Wait()

	// ---- 中心节点：合并 ----
	globalTD := tdigest.NewWithCompression(100)
	for _, inst := range instances {
		globalTD.Merge(inst.td)
		fmt.Printf("  Pod %d: local P99=%.1fms\n", inst.id, inst.td.Quantile(0.99))
	}

	fmt.Println("\n🌐 t-digest 全局分位数:")
	fmt.Printf("  P50:  %.1fms\n", globalTD.Quantile(0.50))
	fmt.Printf("  P90:  %.1fms\n", globalTD.Quantile(0.90))
	fmt.Printf("  P99:  %.1fms\n", globalTD.Quantile(0.99))
	fmt.Printf("  P999: %.1fms\n", globalTD.Quantile(0.999))
}

简单说，“多个实例的 Stream” 就是指：同一个服务的多个进程（或容器/Pod），各自内存里都有一个 
"quantile.Stream" 对象，分别统计自己那部分流量的延迟分布。

用一个具体场景理解

假设你有一个订单 API，部署了 3 个实例（Pod A、B、C），前面挂了负载均衡：

用户请求 → 负载均衡 → ┬─ Pod A (Stream_A)
                      ├─ Pod B (Stream_B)
                      └─ Pod C (Stream_C)

每个 Pod 里都跑着这样的代码：

var stream = quantile.NewTargeted(map[float64]float64{
    0.99: 0.001,
})

// 每个请求进来就 Insert
stream.Insert(latencyMs)

问题来了

- Pod A 这分钟处理了 1000 个请求，算出 P99 = 80ms
- Pod B 处理了 1000 个请求，算出 P99 = 120ms
- Pod C 处理了 1000 个请求，算出 P99 = 300ms（刚好碰到一批慢查询）

❓ 全局 P99 是多少？
你不能简单取平均值 
"(80+120+300)/3 = 166ms"，因为分位数不能这么算。

你也不能只看 Pod C 的 300ms，因为那只是 1/3 的流量。

真正的全局 P99 应该是：把 3000 个请求的延迟全部混在一起排序，取第 99 百分位的值。

为什么“多个实例的 Stream”是个难题？

方案 1：把每个 Pod 的原始数据都传回中心节点？

❌ 不行。3000 个请求还好，如果是每秒 100 万请求，网络和处理成本爆炸。

方案 2：把每个 Pod 的 Stream 直接 Merge？

⚠️ perks 支持，但有坑：

// 中心节点
globalStream := quantile.NewTargeted(map[float64]float64{0.99: 0.001})
globalStream.Merge(streamA.Samples())
globalStream.Merge(streamB.Samples())
globalStream.Merge(streamC.Samples())

fmt.Println(globalStream.Query(0.99))

问题：

- perks 的 
"Merge" 实现不保证精度，合并后误差会叠加
- 每个 Stream 内部已经做了数据压缩（只保留摘要点），合并时信息已经丢失了一部分
- 实例越多，误差越大

正确的多实例聚合思路

✅ 方案 A：Histogram + Prometheus（最常用）

每个 Pod 不存 Stream，而是把延迟分桶：

// 每个 Pod 暴露指标
histogram := prometheus.NewHistogram(prometheus.HistogramOpts{
    Name:    "http_request_duration_ms",
    Buckets: []float64{10, 25, 50, 100, 250, 500, 1000},
})

histogram.Observe(latencyMs)

Prometheus 服务端用 
"histogram_quantile(0.99, rate(...))" 在查询时聚合所有 Pod 的桶，算出全局 P99。

✅ 优点：任意分位随时查、误差可控、天然支持多实例
❌ 缺点：需要 Prometheus 生态

✅ 方案 B：t-digest 跨节点合并

t-digest 算法专门为合并设计，合并后误差仍然可控：

// 每个 Pod
td := tdigest.NewWithCompression(100)
td.Add(latencyMs)

// 中心节点合并
global := tdigest.NewWithCompression(100)
global.Merge(td1)
global.Merge(td2)
global.Merge(td3)

global.Quantile(0.99)

✅ 优点：流式 + 可合并 + 尾部精度好
❌ 缺点：需要引入 t-digest 库

一句话总结

场景 用什么
单进程内统计 
"perks/quantile" Stream ✅
多实例 + 已有 Prometheus Histogram ✅
多实例 + 无 Prometheus + 流式 t-digest ✅
多实例硬用 perks Merge ⚠️ 能用但误差不可控
